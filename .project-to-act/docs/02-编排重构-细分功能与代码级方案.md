## 2. 运行状态 State v2（state.py〔新〕）

### 2.1 完整 schema

```python
from typing import Annotated, Any, Literal, TypedDict
from langgraph.graph.message import add_messages

Route = Literal["workflow", "free", "chat"]
RunStatus = Literal["running","waiting_human","completed","failed","blocked","cancelled"]

class StepRecord(TypedDict, total=False):
    step_id: str; kind: Literal["tool","skill"]; name: str
    status: Literal["pending","ready","running","completed","failed","blocked","skipped","stale"]
    depends_on: list[str]
    attempt: int
    result_summary: dict; evidence_ref: list[str]; trace_id: str
    started_at: str; finished_at: str; error: dict | None

class RouteDecision(TypedDict, total=False):
    source: Literal["explicit","llm","deterministic_fallback"]
    route: Route; intent: str; workflow_id: str | None
    tools: list[str]; skills: list[str]
    confidence: float; reason: str; raw: dict | None   # raw=LLM 原始输出（审计）

class RunStateV2(TypedDict, total=False):
    # ── 标识 ──
    run_id: str; task_id: str; tenant_id: str; thread_id: str
    # ── 输入 ──
    message: str
    request: dict                      # 原始请求（显式 workflow/tools/skills/参数）
    attachments: list[dict]            # {name,kind,content_b64,sha256,sniffed_format}
    # ── 会话记忆（痛点 11 修复：真实读写）──
    messages: Annotated[list[Any], add_messages]
    # ── 路由与计划 ──
    intent: str; route: Route; route_decision: RouteDecision
    workflow_id: str | None
    plan: list[StepRecord]
    # ── 执行 ──
    outputs: dict[str, dict]           # step_id → Goal 信封
    evidence: list[dict]               # 全链证据（沿旧 evidence 结构）
    steps: list[dict]                  # 步骤事件流（沿旧 StepRecord 摘要，审计用）
    errors: list[dict]
    retry_counts: dict[str, int]       # step_id → 已重试次数（上限 config.max_step_retries）
    # ── 审查与门 ──
    pending_gate: dict | None          # {gate_type, tool, payload_digest, allowed_roles, reason}
    approvals: list[dict]              # {gate_type, decision, actor, role, at, note}
    authorized_steps: list[str]
    review_findings: list[dict]        # 审查 Agent 的规则命中记录（可审计）
    # ── 知识（进化闭环）──
    knowledge_context: list[dict]      # 注入的活跃知识（top-k）
    knowledge_consumed: bool           # 统筹是否真实消费（进化侧命中率统计）
    # ── 终态 ──
    status: RunStatus
    summary: dict
    trace: list[dict]; model: dict
```

### 3.1 build_graph 完整设计

```python
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.sqlite.aio import AsyncSqliteSaver

def build_graph(deps: "GraphDeps") -> "CompiledStateGraph":
    g = StateGraph(RunStateV2)
    g.add_node("orchestrator_plan", orchestrator_plan_node(deps))     # 统筹：路由+计划+注入消费
    g.add_node("orchestrator_dispatch", orchestrator_dispatch_node(deps))  # 统筹：选就绪步骤
    g.add_node("worker_execute", worker_execute_node(deps))           # Worker：装配+调用
    g.add_node("reviewer_check", reviewer_check_node(deps))           # 审查：规则+Gate(interrupt)
    g.add_node("chat_answer", chat_answer_node(deps))                 # chat 路径
    g.add_node("finalize", finalize_node(deps))                       # 终态汇总
    g.add_node("evolution_observe", evolution_observe_node(deps))     # 进化观察（吞异常）

    g.add_edge(START, "orchestrator_plan")
    g.add_conditional_edges("orchestrator_plan", route_after_plan, {
        "execute": "orchestrator_dispatch",
        "chat": "chat_answer",
        "finalize": "finalize",
    })
    g.add_edge("orchestrator_dispatch", "worker_execute")
    g.add_edge("worker_execute", "reviewer_check")
    g.add_conditional_edges("reviewer_check", route_after_review, {
        "next":   "orchestrator_dispatch",   # 通过且还有就绪/待就绪步骤
        "retry":  "worker_execute",          # 可修复失败且 retry_counts 未超限（统筹修正 payload 后重试）
        "replan": "orchestrator_plan",       # BLOCKED_INPUT 需补数(data Gate)或计划级调整
        "finalize": "finalize",              # 全部完成 / 不可修复失败
    })
    g.add_edge("chat_answer", "evolution_observe")
    g.add_edge("finalize", "evolution_observe")
    g.add_edge("evolution_observe", END)

    return g.compile(checkpointer=deps.checkpointer)   # 唯一编译点；Studio 与生产同图
```

## 4. 统筹 Agent（orchestrator/）

### 4.1 router.py〔新〕——路由决策器

```python
@dataclass(frozen=True)
class RouteDecision_: ...   # 即 §2.1 RouteDecision 的运行时对象

class Router:
    def __init__(self, llm: LLMClient, tools: "ToolRegistry", skills: "SkillRegistry",
                 workflows: "WorkflowEngine"): ...

    def visible_catalog(self) -> dict:
        """喂给 LLM 的目录：只含 BOUND_LOCAL∪BOUND_HTTP 工具（名称+描述截断 80 字），
        已注册技能（含 usage_order 提示），KNOWN_WORKFLOWS（id+一句话）。UNBOUND/SANDBOX 不进。"""

    async def decide(self, state: RunStateV2) -> RouteDecision:
        if d := self._explicit(state["request"]):            # 1) 显式一票否决
            return d
        try:
            proposal = await self.llm.route(                 # 2) LLM 提案
                message=state["message"],
                history=state["messages"][-8:],              # 记忆窗口
                attachments_meta=[{k: a.get(k) for k in ("name","kind","sniffed_format")}
                                   for a in state.get("attachments", [])],
                catalog=self.visible_catalog(),
                knowledge=state.get("knowledge_context", []),  # 进化注入（缺口②修复点）
            )
            if d := self._validate(proposal):                #    route 枚举/workflow 已知/工具已绑定/技能偏序
                return d
        except LLMError:
            pass
        return self._fallback(state)                          # 3) 最小确定性回退
```

**路由 prompt（prompts.py，全文设计）**：

```
你是云湃制造系统的统筹路由器。只识别意图与选择执行路径，绝不执行工具、绝不编造数值。
必须只输出一个 JSON 对象：{"intent": str, "route": "workflow"|"free"|"chat",
"workflow_id": str|null, "tools": [str], "skills": [str], "confidence": 0~1, "reason": str}
规则：
1) route 只有三个合法值：workflow（完整受控主链）、free（组合已列出的工具/技能）、chat（解释性对话）。
2) workflow_id 只能从【可用工作流】中选；tools 只能从【已绑定工具】中选；skills 只能从【可用技能】中选。
```

#### 6.2.1 写工具三段式契约（propose → approve → commit）〔新，D-005〕

> **来源**：账本 B-001（P0）裁定 —— 2026-09-13 用户决策 D-005；**范围收窄为 M6 专属 —— 2026-09-14 用户决策 D-010**。本条文为 **M6 写工具**的代码级唯一依据；**不构成对 M0–M5 的强制**，存量写工具维持既有"执行后开门"语义（B-014 据 D-010 已关闭）。

**背景**：v2 图内**唯一开门点**是 `reviewer_check_node`（`graph.py:317`），而它在 `worker_execute` **之后**执行（`interrupt` 在 `:362`，批准后的副作用钩子在 `:188` / `:269` / `:156`）。因此当前语义是"工具已经把库写了，然后才开门等人工批"。对财务模块（成本、月账、单据）不可接受。

**契约（限 M6 写工具）**：M6 中**有财务后果**的写操作，其"生效"必须发生在人工门批准**之后**。（这是一个可复制的**模式**：别的模块若出现"生效事实不得先落库"的同类需求，可照此实现；但本条文**不推广义务**——见下"范围"。）

| 段 | 执行者 | 允许 | 禁止 |
|---|---|---|---|
| **propose** | Worker（工具执行阶段） | 产出草稿/提案：`status=draft`／`trial` 的行、或纯计算输出 | **不得**产生任何有财务/事实后果的落库（不进汇总、不构成生效事实） |
| **approve** | 审查 Agent + 人工 | 开门并 `interrupt` 挂起；人工 `approve`／`reject` | LLM 置信度任何阈值都不得自动 approve（沿用 §6.2 红线） |
| **commit** | `_apply_*` 钩子（resume 后） | 翻正状态／冻结／发布，写回 trace 与 `review_applied` | `reject` 路径不得落任何生效行 |

**门型**：财务确认／月末结账／单据落库用 **`finance`**（角色 `finance-officer` / `admin`，合法决策 `approve` / `reject`），与 `GATE_ALLOWED_ROLES` / `GATE_DECISIONS` 同表登记。

**落地范式**：commit 段照既有 `_apply_m5_release`（发布 + head CAS，冲突不抛异常而是写回 `pending_gate` 保持可恢复）与 `_apply_candidate_approval` 的写法，**不改 `reviewer_check_node` 结构**，仅在 approve 分支追加一行分派。

**范围（M6 专属，D-010）**：本条文**只约束 M6 写工具**，不构成对 M0–M5 的强制。M0 6 类 / M1 4 件 / M3 4 件 / M4 10 件 / M5 5 件写工具**维持既有"执行后开门"语义**——其领域不要求"生效事实后置"，**不计为设计债**（账本 B-014 已据 D-010 关闭）。
